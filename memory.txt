    // EXAMPLE 1

    let myHouse1 = { color: 'blue' };
    // myHouse1 : [ m1 ]
    
    //
    // m1 
    // {
    //  color: [ 'blue' ]
    // }

    let myHouse2 = myHouse1
    // myHouse2 : [ m1 ]
    
    myHouse2.color = 'red'
    // m1
    // {
    //  color: [ 'red' ]
    // }

    // EXAMPLE 2 

    let computer = { memory: { type: "RAM", capacity: 64}, isOn: false }

    // computer : [ m1 ]

    // m1 {
    //   memory: [ m2 ]
    //    isOn: [ false ]
    // }

    // m2 {
    //  type: [ "RAM" ]
    //  capacity: [  64  ]
    // }

    let memory = computer.memory;

    // memory : [ m2 ]

    memory.capacity = memory.capacity / 2;

    // m2.capacity : [ 64 / 2 ]
    // m2.capacity : [ 32 ]

    // m2 {
    //  type: [ "RAM" ]
    //  capacity: [  32  ]
    // }

    // EXAMPLE 3 

    var myArray = ['a', 'b', {}]
    // myArray : [ m1 ]

    // m1 {
    //  0: [ 'a' ]
    //  1: [ 'b' ]
    //  2: [ m2 ]
    // }

    // m2 = {}

    myArray[0] = 'g'

    // m1 {
    //  0: [ 'g' ] - 'a' becomes 'g'
    //  1: [ 'b' ]
    //  2: [ m2 ]
    // }

    myArray[myArray.length] = ['x'];

    // m1 = [
    //  0: [ 'g' ]
    //  1: [ 'b' ]
    //  2: [ m2 ]
    //  3: [ m3 ]
    // ]

    // m3 : [ 'x' ]



    // Example 4  Warm Up


    let a = {x: 1, y: {z: true}};
    // a: [ m1 ]

    // m1 {
    //  x: [ 1 ],
    //  y: [ m2 ]
    // }

    // m2 = {
    //  z: [ true ] 
    // }



    let b = a.y;
    // b : [ m2 ]

    b.z = false;

    // m2 {
    //  z: [ false ] 
    // }


    b = {};

    // b : [ m3 ]

    // m3 {}

    // a = {x: 1, y: {z: false}};



    // EXAMPLE 5 

    let weather = { temperature: { degrees: 81, unit: "F"}, isCloudy: false };
    // weather : [ m1 ]

    // m1 {
    //  temperature : [ m2 ],
    //  isCloudy : [ false ]
    // }

    // m2 {
    //  degrees : [ 81 ],
    //  unit : [ "F" ]
    // }


    let temperature = weather.temperature;
    // temperature : [ m2 ]

    temperature.degrees = temperature.degrees - 10;
    // m2 {
    //  degrees = [ 81 - 10 ],
    //  ...
    // }

    // m2 {
    //  degrees = [ 71 ],
    //  ...
    // }

    temperature = {};
    // temperature : [ m3 ]
    
    // m3 {} 


    temperature.degrees = 4;
    // m3 {
    //  degrees: 4
    // }
